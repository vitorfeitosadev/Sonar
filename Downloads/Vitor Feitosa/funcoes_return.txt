def soma(a,b):
    return a+b
def sub(a,b):
    return a-b
def mult(a,b):
    return a*b
def div(a,b):
    return a/b
def resto_div(a,b):
    return a%b

def calculadora(a, b, operacao):
    if operacao == 1:
        return soma(a,b)
    elif operacao == 2:
        return sub(a,b)
    elif operacao == 3:
        return mult(a,b)
    elif operacao == 4:
        return div(a,b)
    elif operacao == 5:
        return resto_div(a,b)
    else: 
        print("Operacao invalida!")

while True:
    print(calculadora(float(input("digite um numero: ")),float(input("digite outro numero: ")),int(input("qual operação você quer fazer:\n1 para soma\n2 para subtracao \n3 para multiplicação\n4 para divisão\n5 para resto de divisão\n -> "))))
    repetir = (input("quer continuar calculando? (y/n)").lower())
    if repetir != "y":
        break
